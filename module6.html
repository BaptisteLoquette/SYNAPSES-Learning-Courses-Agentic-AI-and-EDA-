Python — Stage A: Identification Agent Implementation 
from langchain_anthropic import ChatAnthropic
from langchain.output_parsers import PydanticOutputParser
from pydantic import BaseModel, Field
from typing import List, Optional

# ── Output Schema (Pydantic) ──────────────────────────────────
class ComponentSpec(BaseModel):
    type: str = Field(description="Component type: NMOS, PMOS, capacitor, resistor, etc.")
    role: str = Field(description="Functional role in the circuit")
    quantity: int = Field(default=1)

class CircuitInventory(BaseModel):
    circuit_type: str = Field(description="OTA, ADC, LNA, etc.")
    topology: str = Field(description="Specific circuit topology name")
    pdk: str = Field(description="Target process design kit: sky130, gf180, tsmc28")
    specs: dict = Field(description="Performance specifications as key-value pairs")
    components: List[ComponentSpec] = Field(description="Required components list")
    constraints: Optional[List[str]] = Field(default=[], description="Design constraints")

# ── Identification Agent ──────────────────────────────────────
class IdentificationAgent:
    def __init__(self):
        self.llm = ChatAnthropic(model="claude-opus-4-5")
        self.parser = PydanticOutputParser(pydantic_object=CircuitInventory)
    
    def identify(self, natural_language_spec: str) -> CircuitInventory:
        """Parse NL spec into structured component inventory."""
        
        prompt = f"""You are an expert analog IC designer. Parse the following natural 
language circuit specification into a structured component inventory.

Specification: {natural_language_spec}

Extract:
1. Circuit type and topology
2. All performance specifications with units
3. Required components and their functional roles
4. Any design constraints mentioned
5. Target process (default to sky130 if not specified)

{self.parser.get_format_instructions()}"""
        
        response = self.llm.invoke(prompt)
        inventory = self.parser.parse(response.content)
        
        # Validate extracted specs make physical sense
        self._validate_inventory(inventory)
        
        return inventory
    
    def _validate_inventory(self, inv: CircuitInventory):
        """Basic sanity checks on extracted specs."""
        specs = inv.specs
        
        if "gain_dB" in specs and specs["gain_dB"] > 120:
            raise ValueError(f"Gain {specs['gain_dB']} dB is unrealistic for a single-path analog circuit")
        
        if "phase_margin_deg" in specs and specs["phase_margin_deg"] < 30:
            import warnings
            warnings.warn(f"Phase margin {specs['phase_margin_deg']}° below 45° will result in ringing")
        
        if "power_budget_mW" in specs and specs["power_budget_mW"] < 0.001:
            raise ValueError("Power budget below 1µW is unrealistic for active analog circuits")

Stage B: Feedback-Enhanced Netlist Generator

STAGE
B
Feedback-Enhanced Netlist Generator
Generate and iteratively refine SPICE netlists using simulation feedback
Objective

Build a closed-loop that generates a SPICE netlist using AnalogCoder principles, runs simulation, and iteratively improves until PPA targets are met.

Tools Used
PySpice LangGraph Ngspice SKY130 PDK
Loop Logic
  1. LLM generates initial PySpice code
  2. Execute code → SPICE netlist
  3. Run Ngspice simulation
  4. Extract metrics (gain, GBW, PM)
  5. Compare to PPA targets
  6. If fails → LLM refines with error context
Python — Stage B: Netlist Generator with Feedback Loop 
from langgraph.graph import StateGraph, END
from langgraph.checkpoint.sqlite import SqliteSaver
from langchain_anthropic import ChatAnthropic
from typing import TypedDict, List, Optional, Annotated
import operator, subprocess, tempfile, os

class NetlistState(TypedDict):
    inventory: dict          # From Stage A
    pyspice_code: str        # Generated Python code
    spice_netlist: str       # Executed output
    sim_results: dict        # Ngspice output
    ppa_achieved: dict       # Extracted metrics
    ppa_targets: dict        # From inventory
    errors: Annotated[List[str], operator.add]
    iteration: int
    max_iterations: int

llm = ChatAnthropic(model="claude-opus-4-5")

def generate_netlist_code(state: NetlistState) -> dict:
    """Generate PySpice code for the circuit."""
    
    previous_error = state["errors"][-1] if state["errors"] else "None"
    iteration = state["iteration"]
    
    prompt = f"""Generate PySpice Python code for this analog circuit:

Circuit inventory: {state['inventory']}
Iteration: {iteration}
Previous error/feedback: {previous_error}

Generate complete PySpice code that:
1. Creates the circuit using PySpice netlist API
2. Includes all components from the inventory
3. Uses SKY130 MOSFET models (NMOS_SKY130, PMOS_SKY130)
4. Targets: Gain={state['ppa_targets'].get('gain_dB', 60)}dB, 
           GBW={state['ppa_targets'].get('GBW_MHz', 10)}MHz,
           PM>{state['ppa_targets'].get('phase_margin_deg', 60)}°
5. Includes .ac analysis from 1Hz to 1GHz

Output ONLY executable Python code."""
    
    response = llm.invoke(prompt)
    code = extract_python_code(response.content)
    return {"pyspice_code": code, "iteration": iteration + 1}

def execute_and_simulate(state: NetlistState) -> dict:
    """Execute PySpice code and run Ngspice."""
    
    # Execute the PySpice code to get SPICE netlist
    namespace = {}
    try:
        exec(state["pyspice_code"], {"__name__": "__main__"}, namespace)
        circuit = namespace.get("circuit")
        if not circuit:
            return {"errors": ["PySpice code did not create a 'circuit' object"]}
        spice_netlist = str(circuit)
    except Exception as e:
        return {"errors": [f"PySpice execution error: {e}"]}
    
    # Run Ngspice simulation
    with tempfile.NamedTemporaryFile(suffix=".sp", mode="w", delete=False) as f:
        f.write(spice_netlist)
        netlist_path = f.name
    
    result = subprocess.run(
        ["ngspice", "-b", "-o", "/tmp/sim.txt", netlist_path],
        capture_output=True, text=True, timeout=120
    )
    os.unlink(netlist_path)
    
    if result.returncode != 0:
        return {
            "spice_netlist": spice_netlist,
            "errors": [f"Ngspice failed: {result.stderr[:500]}"]
        }
    
    # Parse simulation results
    sim_data = parse_ngspice_ac_results("/tmp/sim.txt")
    ppa = extract_ppa_metrics(sim_data)
    
    return {
        "spice_netlist": spice_netlist,
        "sim_results": sim_data,
        "ppa_achieved": ppa
    }

def check_ppa(state: NetlistState) -> str:
    """Routing: met targets? → Stage C. Max iter? → DONE. Else → retry."""
    achieved = state["ppa_achieved"]
    targets = state["ppa_targets"]
    
    all_met = all(achieved.get(k, 0) >= v for k, v in targets.items())
    
    if all_met:
        return "success"
    elif state["iteration"] >= state["max_iterations"]:
        return "max_iterations"
    else:
        return "retry"

# Build Stage B graph
stage_b = StateGraph(NetlistState)
stage_b.add_node("generate", generate_netlist_code)
stage_b.add_node("simulate", execute_and_simulate)
stage_b.set_entry_point("generate")
stage_b.add_edge("generate", "simulate")
stage_b.add_conditional_edges("simulate", check_ppa, {
    "success": END,
    "retry": "generate",
    "max_iterations": END
})

checkpointer = SqliteSaver.from_conn_string("./stage_b_checkpoint.db")
stage_b_app = stage_b.compile(checkpointer=checkpointer)

Stage C: Multimodal Waveform Debugger

STAGE
C
Multimodal Waveform Debugger
Implement an agent that "looks at" output waveforms using a vision LLM
Objective

Build an agent that generates a PNG of the Bode plot, identifies instability or spec violations visually, and suggests specific parameter changes (bias currents, compensation values).

Vision Models
Claude 4.5 Vision GPT-5 Vision Gemini 3 Pro
What the Agent Looks For
  • DC gain value (flat region)
  • GBW (0dB crossover frequency)
  • Phase margin at GBW
  • Gain peaking (instability)
  • Second pole location
  • Unity gain stability
Python — Stage C: Multimodal Waveform Debugger 
import base64
import json
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import numpy as np
from langchain_anthropic import ChatAnthropic
from langchain_core.messages import HumanMessage
from pydantic import BaseModel, Field
from typing import List, Optional

class BodeAnalysis(BaseModel):
    dc_gain_dB: float = Field(description="DC gain extracted from plot")
    gwb_MHz: float = Field(description="Gain-bandwidth product in MHz")
    phase_margin_deg: float = Field(description="Phase margin at GBW crossing")
    has_gain_peaking: bool = Field(description="Whether gain peaking is visible")
    instability_risk: str = Field(description="low/medium/high")
    parameter_suggestions: List[str] = Field(description="Specific parameter changes to improve")

class WaveformDebugAgent:
    
    def __init__(self):
        self.vision_llm = ChatAnthropic(model="claude-opus-4-5")
    
    def generate_bode_png(self, frequencies: np.ndarray, 
                           gain_dB: np.ndarray, phase_deg: np.ndarray) -> bytes:
        """Generate a publication-quality Bode plot PNG."""
        fig, axes = plt.subplots(2, 1, figsize=(10, 8))
        fig.patch.set_facecolor('#0a0e1a')
        
        for ax in axes:
            ax.set_facecolor('#141d35')
            ax.tick_params(colors='#94a3b8')
            ax.xaxis.label.set_color('#94a3b8')
            ax.yaxis.label.set_color('#94a3b8')
            for spine in ax.spines.values():
                spine.set_edgecolor('#2d3748')
        
        # Gain plot
        axes[0].semilogx(frequencies, gain_dB, '#60a5fa', linewidth=2.5)
        axes[0].axhline(y=0, color='#ef4444', linestyle='--', alpha=0.7, label='0 dB')
        axes[0].axhline(y=3, color='#fbbf24', linestyle=':', alpha=0.5, label='+3 dB')
        axes[0].set_ylabel('Gain (dB)', color='#f0f4ff')
        axes[0].set_title('Bode Plot Analysis', color='#f0f4ff', fontweight='bold')
        axes[0].legend(facecolor='#1a2540', edgecolor='#2d3748', labelcolor='#94a3b8')
        axes[0].grid(True, which='both', alpha=0.2, color='#2d3748')
        
        # Phase plot
        axes[1].semilogx(frequencies, phase_deg, '#10b981', linewidth=2.5)
        axes[1].axhline(y=-180, color='#ef4444', linestyle='--', alpha=0.7, label='-180°')
        axes[1].axhline(y=-135, color='#fbbf24', linestyle=':', alpha=0.5, label='-135° (45° PM)')
        axes[1].set_ylabel('Phase (degrees)', color='#f0f4ff')
        axes[1].set_xlabel('Frequency (Hz)', color='#f0f4ff')
        axes[1].legend(facecolor='#1a2540', edgecolor='#2d3748', labelcolor='#94a3b8')
        axes[1].grid(True, which='both', alpha=0.2, color='#2d3748')
        
        plt.tight_layout()
        
        import io
        buf = io.BytesIO()
        plt.savefig(buf, format='png', dpi=150, bbox_inches='tight',
                    facecolor='#0a0e1a')
        plt.close()
        buf.seek(0)
        return buf.read()
    
    def analyze_visually(self, frequencies: np.ndarray,
                          gain_dB: np.ndarray, phase_deg: np.ndarray,
                          targets: dict) -> BodeAnalysis:
        """Analyze Bode plot using vision LLM."""
        
        # Generate the plot
        png_bytes = self.generate_bode_png(frequencies, gain_dB, phase_deg)
        image_b64 = base64.b64encode(png_bytes).decode('utf-8')
        
        analysis_prompt = f"""You are analyzing a Bode plot for a two-stage Miller OTA.

Design targets:
- DC Gain: ≥ {targets.get('gain_dB', 60)} dB
- GBW: ≥ {targets.get('GBW_MHz', 10)} MHz  
- Phase Margin: ≥ {targets.get('phase_margin_deg', 60)}°

Carefully examine the Bode plot and provide:
1. Measured DC gain (flat region value in dB)
2. GBW — frequency where gain crosses 0 dB
3. Phase at the GBW crossing frequency → phase margin = 180° + phase_at_GBW
4. Whether you observe gain peaking (instability warning)
5. Instability risk assessment: low/medium/high
6. Specific circuit parameter changes to improve performance:
   - If phase margin < 60°: suggest Miller capacitor adjustment
   - If GBW too low: suggest bias current or W/L ratio changes
   - If DC gain too low: suggest adding a cascode stage

Output as JSON matching the BodeAnalysis schema."""
        
        response = self.vision_llm.invoke([HumanMessage(content=[
            {"type": "image", "source": {
                "type": "base64", "media_type": "image/png", "data": image_b64
            }},
            {"type": "text", "text": analysis_prompt}
        ])])
        
        # Parse JSON from response
        analysis_text = response.content
        data = json.loads(extract_json(analysis_text))
        return BodeAnalysis(**data)

# ── Usage in the agentic workflow ─────────────────────────────
debugger = WaveformDebugAgent()
analysis = debugger.analyze_visually(frequencies, gain_dB, phase_deg, targets)

print(f"DC Gain: {analysis.dc_gain_dB:.1f} dB")
print(f"GBW: {analysis.gwb_MHz:.1f} MHz")
print(f"Phase Margin: {analysis.phase_margin_deg:.1f}°")
print(f"Instability Risk: {analysis.instability_risk}")
print("Suggested improvements:")
for suggestion in analysis.parameter_suggestions:
    print(f"  → {suggestion}")

Stage D: Autonomous Layout Synthesis

STAGE
D
Autonomous Layout Synthesis
Automate physical design with DRC/LVS verification and iterative correction
Objective

Use ALIGN or OpenROAD to generate physical layout from the verified netlist. The agent uses a "proofreading strategy" to check DRC/LVS results and adjust placement constraints iteratively.

Tools
ALIGN OpenROAD KLayout DRC
Proofreading Strategy
  1. Generate initial layout with ALIGN
  2. Run KLayout DRC check
  3. Parse DRC violations
  4. LLM analyzes violations → suggests constraint edits
  5. Update placement constraints file
  6. Re-run ALIGN with new constraints
Python — Stage D: Layout Synthesis with Proofreading 
import subprocess
import json
from langchain_anthropic import ChatAnthropic
from typing import TypedDict, List, Annotated
import operator

class LayoutState(TypedDict):
    spice_netlist: str
    constraints: dict          # ALIGN constraint file content
    gds_file: str              # Path to generated layout
    drc_violations: List[dict]
    lvs_status: str
    errors: Annotated[List[str], operator.add]
    iteration: int

llm = ChatAnthropic(model="claude-opus-4-5")

def generate_align_constraints(state: LayoutState) -> dict:
    """LLM generates ALIGN constraint file for analog layout."""
    
    prompt = f"""Generate ALIGN constraints for this OTA netlist.
    
Netlist summary: {summarize_netlist(state['spice_netlist'])}
Previous DRC violations: {state.get('drc_violations', [])}

Generate JSON constraints for ALIGN specifying:
1. Symmetry constraints (matched transistor pairs must be symmetric)
2. Ordering constraints (place tail current below diff pair)  
3. Guard ring requirements
4. WPE mitigation: set min_well_distance for all NMOS devices
5. Interdigitation for M3/M4 current mirror matching

Key rules:
- Differential pair M1/M2 must have SYMMETRIC constraint
- All matched pairs must use INTERDIGITATED placement
- Set GUARD_RING: true for all sensitive nodes

Output valid ALIGN JSON constraint format."""
    
    response = llm.invoke(prompt)
    constraints = json.loads(extract_json(response.content))
    return {"constraints": constraints}

def run_align_layout(state: LayoutState) -> dict:
    """Execute ALIGN to generate layout from netlist + constraints."""
    import tempfile, os
    
    # Write netlist and constraints to temp directory
    with tempfile.TemporaryDirectory() as tmpdir:
        netlist_path = f"{tmpdir}/ota.sp"
        constraint_path = f"{tmpdir}/ota.const.json"
        
        with open(netlist_path, "w") as f:
            f.write(state["spice_netlist"])
        with open(constraint_path, "w") as f:
            json.dump(state["constraints"], f)
        
        # Run ALIGN
        result = subprocess.run([
            "python", "-m", "align",
            "--pdk_dir", "/pdk/sky130",
            "--netlist", netlist_path,
            "--constraint", constraint_path,
            "--output_dir", tmpdir
        ], capture_output=True, text=True, timeout=600)
        
        if result.returncode != 0:
            return {"errors": [f"ALIGN failed: {result.stderr[:500]}"]}
        
        gds_path = f"{tmpdir}/ota.gds"
        if not os.path.exists(gds_path):
            return {"errors": ["ALIGN did not produce GDS output"]}
        
        # Copy GDS to permanent location
        import shutil
        final_gds = f"./outputs/ota_iter{state['iteration']}.gds"
        shutil.copy(gds_path, final_gds)
        return {"gds_file": final_gds}

def run_drc(state: LayoutState) -> dict:
    """Run KLayout DRC on generated layout."""
    result = subprocess.run([
        "klayout", "-b", "-r", "/pdk/sky130/sky130.lydrc",
        "-rd", f"input={state['gds_file']}",
        "-rd", "output=/tmp/drc_results.xml"
    ], capture_output=True, text=True, timeout=300)
    
    violations = parse_drc_xml("/tmp/drc_results.xml")
    return {
        "drc_violations": violations,
        "lvs_status": "pending" if violations else "drc_clean"
    }

def analyze_and_fix_violations(state: LayoutState) -> dict:
    """LLM analyzes DRC violations and suggests constraint updates."""
    
    if not state["drc_violations"]:
        return {}  # No violations to fix
    
    prompt = f"""Analyze these DRC violations and update the ALIGN constraints to fix them.

DRC violations:
{json.dumps(state['drc_violations'][:10], indent=2)}

Current constraints:
{json.dumps(state['constraints'], indent=2)}

For each violation:
1. Identify the root cause (insufficient spacing, missing guard ring, etc.)
2. Specify which constraint needs updating
3. Provide the corrected constraint value

Return the complete updated constraint JSON."""
    
    response = llm.invoke(prompt)
    updated_constraints = json.loads(extract_json(response.content))
    return {"constraints": updated_constraints, "iteration": state["iteration"] + 1}

def check_drc_routing(state: LayoutState) -> str:
    if not state["drc_violations"]:
        return "drc_clean"
    elif state["iteration"] >= 5:
        return "max_attempts"
    return "fix_and_retry"

from langgraph.graph import StateGraph, END
from langgraph.checkpoint.postgres import PostgresSaver
from typing import TypedDict, List, Optional, Annotated
import operator

class FullEDAState(TypedDict):
    """Combined state for all four stages."""
    # Input
    nl_spec: str
    
    # Stage A outputs
    inventory: Optional[dict]
    ppa_targets: dict
    
    # Stage B outputs
    pyspice_code: str
    spice_netlist: str
    sim_results: dict
    ppa_achieved: dict
    
    # Stage C outputs
    bode_analysis: Optional[dict]
    waveform_feedback: List[str]
    
    # Stage D outputs
    layout_constraints: dict
    gds_file: Optional[str]
    drc_clean: bool
    
    # Control
    errors: Annotated[List[str], operator.add]
    stage_a_iterations: int
    stage_b_iterations: int
    stage_d_iterations: int

# ── Full Pipeline Graph ───────────────────────────────────────
pipeline = StateGraph(FullEDAState)

# Add all nodes
pipeline.add_node("identify",       identification_agent_node)    # Stage A
pipeline.add_node("generate",       generate_netlist_code)        # Stage B  
pipeline.add_node("simulate",       execute_and_simulate)         # Stage B
pipeline.add_node("debug_waveform", waveform_debug_node)          # Stage C
pipeline.add_node("gen_constraints",generate_align_constraints)   # Stage D
pipeline.add_node("layout",         run_align_layout)             # Stage D
pipeline.add_node("drc",            run_drc)                      # Stage D
pipeline.add_node("fix_violations", analyze_and_fix_violations)   # Stage D

# Define flow
pipeline.set_entry_point("identify")
pipeline.add_edge("identify", "generate")
pipeline.add_edge("generate", "simulate")

# Stage B conditional: sim pass/fail
pipeline.add_conditional_edges("simulate", check_ppa, {
    "success": "debug_waveform",
    "retry": "generate",
    "max_iterations": END
})

# Stage C: visual analysis → possible return to Stage B
pipeline.add_conditional_edges("debug_waveform", check_visual_analysis, {
    "approved": "gen_constraints",
    "needs_refinement": "generate"  # Back to Stage B with visual feedback
})

# Stage D: layout → DRC → fix loop
pipeline.add_edge("gen_constraints", "layout")
pipeline.add_edge("layout", "drc")
pipeline.add_conditional_edges("drc", check_drc_routing, {
    "drc_clean": END,
    "fix_and_retry": "fix_violations",
    "max_attempts": END
})
pipeline.add_edge("fix_violations", "layout")

# Compile with checkpointing
checkpointer = PostgresSaver.from_conn_string(
    "postgresql://eda:secret@localhost/eda_workflows"
)
full_pipeline = pipeline.compile(checkpointer=checkpointer,
                                  interrupt_before=["layout"])  # Human approval
